SELECT 
    s.id_stok AS 'ID Mutasi',
    p.id_barang AS 'Kode Barang',
    p.nama_barang AS 'Nama Senjata',
    s.stok_awal AS 'Stok Awal',
    s.jumlah_masuk AS 'Barang Masuk',
    s.jumlah_keluar AS 'Barang Keluar',
    (s.stok_awal + s.jumlah_masuk - s.jumlah_keluar) AS 'Sisa Stok Akhir',
    s.tanggal_mutasi AS 'Waktu Pencatatan',
    s.keterangan AS 'Keterangan Mutasi'
FROM stok s
JOIN produk p ON s.id_barang = p.id_barang
WHERE s.tanggal_mutasi BETWEEN '2026-09-20 00:00:00' AND '2026-09-27 23:59:59'
ORDER BY s.tanggal_mutasi ASC;
